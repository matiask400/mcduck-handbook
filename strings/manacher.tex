\section{Manacher}
\whenuse{Calcular en tiempo lineal el radio del pal\'indromo m\'aximo centrado en
cada posici\'on, para responder existencia/longitud de substrings palindr\'omicos.}

\texttt{odd[i]} es el radio del pal\'indromo impar centrado en $i$, incluyendo
el centro. \texttt{even[i]} es el radio del pal\'indromo par centrado entre
$i-1$ e $i$.

\complexity{$O(n)$ tiempo y memoria.}

\begin{lstlisting}[style=cpp]
pair<vector<int>, vector<int>> manacher(const string &s) {
    int n = s.size();
    vector<int> odd(n), even(n);

    for (int i = 0, left = 0, right = -1; i < n; ++i) {
        int radius = i > right ? 1 : min(odd[left + right - i], right - i + 1);
        while (i - radius >= 0 && i + radius < n &&
               s[i - radius] == s[i + radius]) ++radius;
        odd[i] = radius--;
        if (i + radius > right) left = i - radius, right = i + radius;
    }

    for (int i = 0, left = 0, right = -1; i < n; ++i) {
        int radius = i > right ? 0
                   : min(even[left + right - i + 1], right - i + 1);
        while (i - radius - 1 >= 0 && i + radius < n &&
               s[i - radius - 1] == s[i + radius]) ++radius;
        even[i] = radius--;
        if (i + radius > right) left = i - radius - 1, right = i + radius;
    }
    return {odd, even};
}
\end{lstlisting}

La longitud m\'axima es
$\max(2\cdot\texttt{odd[i]}-1,2\cdot\texttt{even[i]})$.
