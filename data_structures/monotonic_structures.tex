\section{Stack y queue mon\'otonas}
\whenuse{Pr\'oximo mayor/menor, spans, histogramas o m\'inimo/m\'aximo de cada
ventana deslizante en tiempo lineal.}

\begin{lstlisting}[style=cpp]
// Indice del siguiente elemento estrictamente menor; n si no existe.
vector<int> next_smaller(const vector<ll> &a) {
    int n = a.size();
    vector<int> answer(n, n), stack;
    for (int i = 0; i < n; ++i) {
        // Para maximo se cambia por a[i] > a[stack.back()]
        while (!stack.empty() && a[i] < a[stack.back()]) {
            answer[stack.back()] = i;
            stack.pop_back();
        }
        stack.push_back(i);
    }
    return answer;
}

// Minimo de cada ventana de longitud k.
vector<ll> sliding_minimum(const vector<ll> &a, int k) {
    assert(k > 0);
    deque<int> dq;
    vector<ll> answer;
    for (int i = 0; i < (int)a.size(); ++i) {
        while (!dq.empty() && dq.front() <= i - k) dq.pop_front();
        // Para maximo se cambia por a[dq.back()] <= a[i]
        while (!dq.empty() && a[dq.back()] >= a[i]) dq.pop_back();
        dq.push_back(i);
        if (i + 1 >= k) answer.push_back(a[dq.front()]);
    }
    return answer;
}
\end{lstlisting}

\complexity{$O(n)$ tiempo: cada \'indice entra y sale una sola vez; memoria $O(n)$.}
