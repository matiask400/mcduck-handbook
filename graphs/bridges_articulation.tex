\section{Puentes y puntos de articulaci\'on}
\whenuse{Detectar una arista o v\'ertice cuya eliminaci\'on aumenta el n\'umero de
componentes de un grafo no dirigido.}

Versi\'on para multigrafos no dirigidos: identifica la arista padre por ID, no
solamente por v\'ertice.
La DFS usa una pila expl\'icita en memoria din\'amica y admite cadenas largas.

\begin{lstlisting}[style=cpp]
struct LowLink {
    int timer = 0;
    vector<vector<pair<int,int>>> graph; // (to, edge_id)
    vector<int> tin, low;
    vector<char> articulation;
    vector<pair<int,int>> bridges;

    explicit LowLink(int n) : graph(n), tin(n, -1), low(n),
                              articulation(n, false) {}

    void add_edge(int u, int v, int id) {
        graph[u].push_back({v, id});
        graph[v].push_back({u, id});
    }

    void run() {
        struct Frame { int v, parent_edge, next = 0, children = 0; };
        vector<Frame> pending;
        for (int root = 0; root < (int)graph.size(); ++root) {
            if (tin[root] != -1) continue;
            tin[root] = low[root] = timer++;
            pending.push_back({root, -1});
            while (!pending.empty()) {
                auto &frame = pending.back();
                int v = frame.v;
                if (frame.next < (int)graph[v].size()) {
                    auto [to, id] = graph[v][frame.next++];
                    if (id == frame.parent_edge) continue;
                    if (tin[to] != -1) low[v] = min(low[v], tin[to]);
                    else {
                        ++frame.children;
                        tin[to] = low[to] = timer++;
                        pending.push_back({to, id});
                    }
                } else {
                    if (frame.parent_edge == -1 && frame.children > 1)
                        articulation[v] = true;
                    pending.pop_back();
                    if (pending.empty()) continue;
                    int p = pending.back().v;
                    low[p] = min(low[p], low[v]);
                    if (low[v] > tin[p]) bridges.push_back({p, v});
                    if (pending.back().parent_edge != -1 && low[v] >= tin[p])
                        articulation[p] = true;
                }
            }
        }
    }
};

LowLink lowlink(5);

int id = 0;
lowlink.add_edge(0, 1, id++); // 5 veces con los q correspondan conexiones
lowlink.run();
for (auto [u, v] : lowlink.bridges)
    cout << "Puente: " << u << ' ' << v << '\n';
for (int v = 0; v < 5; ++v)
    if (lowlink.articulation[v])
        cout << "Articulacion: " << v << '\n';
\end{lstlisting}

\complexity{$O(V+E)$ tiempo y $O(V+E)$ memoria.}
